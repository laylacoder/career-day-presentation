import pandas as pd

# STAGE 1: Ingest — read raw data from source
raw = pd.read_csv("outreach_records.csv")



# STAGE 1: Transform — clean, normalize, validate
clean = (
  raw
  .dropna(subset=["name", "zip_code", "contacted"])
  .assign(
    zip_code=lambda x: x["zip_code"].astype(str).str.zfill(5),
    state    =lambda x: x["state"].str.upper().str.strip(),
    contacted=lambda x: x["contacted"].str.capitalize()
  )
)



# Aggregate: who hasn't been reached, grouped by zip?
priority = (
  clean[clean["contacted"] == "No"]
  .groupby("zip_code").size()
  .reset_index(name="uncontacted")
  .sort_values("uncontacted", ascending=False)
)



# HAND OFF — write transformed data for downstream
priority.to_parquet("transformed/priority_zips.parquet")
print("✓ Transformed data ready for downstream.")
